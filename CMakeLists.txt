cmake_minimum_required(VERSION 3.22)
project(Distortion VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DISTORTION_BUILD_PLUGIN "Build JUCE plug-ins" ON)
option(DISTORTION_BUILD_UI_TESTS "Build processor and editor integration checks" OFF)
option(DISTORTION_ENABLE_CLAP "Build CLAP wrapper" ON)
set(DISTORTION_SANITIZER "none" CACHE STRING "none, address, thread")
if(MSVC)
  add_compile_options(/utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(DISTORTION_SANITIZER STREQUAL "address")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  elseif(DISTORTION_SANITIZER STREQUAL "thread")
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  endif()
endif()
find_package(Threads REQUIRED)
enable_testing()
add_executable(DistortionDSPTest Tests/dsp_tests.cpp)
target_include_directories(DistortionDSPTest PRIVATE Source)
add_test(NAME DSP COMMAND DistortionDSPTest)
set_tests_properties(DSP PROPERTIES TIMEOUT 240)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/realtime_tests.cpp")
  add_executable(DistortionRealtimeTest Tests/realtime_tests.cpp)
  target_include_directories(DistortionRealtimeTest PRIVATE Source)
  target_link_libraries(DistortionRealtimeTest PRIVATE Threads::Threads)
  add_test(NAME Realtime COMMAND DistortionRealtimeTest)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tools/analyse.cpp")
  add_executable(DistortionAnalyze Tools/analyse.cpp)
  target_include_directories(DistortionAnalyze PRIVATE Source)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME HarmonicCalibration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tools/calibrate.py"
      --analyzer "$<TARGET_FILE:DistortionAnalyze>" --output "${CMAKE_CURRENT_BINARY_DIR}/calibration")
  endif()
endif()
if(NOT DISTORTION_BUILD_PLUGIN)
  return()
endif()
set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/JUCE" CACHE PATH "Path to JUCE 8.0.4 sources")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "JUCE not found. Set JUCE_DIR; see Docs/BUILD.md")
endif()
if(WIN32)
  include(cmake/ApplyWindowsVBlankFix.cmake)
  distortion_apply_windows_vblank_fix("${JUCE_DIR}")
endif()
add_subdirectory("${JUCE_DIR}" JUCE)
set(default_formats VST3)
if(WIN32 OR APPLE)
  list(APPEND default_formats AAX)
endif()
if(APPLE)
  list(APPEND default_formats AU)
endif()
set(DISTORTION_FORMATS "${default_formats}" CACHE STRING "JUCE plug-in formats")
if("AAX" IN_LIST DISTORTION_FORMATS AND AAX_SDK_PATH)
  juce_set_aax_sdk_path("${AAX_SDK_PATH}")
endif()
juce_add_plugin(Distortion
  COMPANY_NAME "Rainline Music" COMPANY_EMAIL "rainlinemusic@gmail.com"
  BUNDLE_ID "ru.rainlinemusic.distortion" AAX_IDENTIFIER "ru.rainlinemusic.distortion"
  PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE DsRn
  FORMATS ${DISTORTION_FORMATS}
  PRODUCT_NAME "Distortion" DESCRIPTION "Sidechain-referenced distortion analyzer"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  VST3_CATEGORIES Fx Analyzer AAX_CATEGORY Effect)
juce_generate_juce_header(Distortion)
target_sources(Distortion PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(Distortion PUBLIC Source)
target_compile_definitions(Distortion PUBLIC
  JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0
  $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
target_link_libraries(Distortion PRIVATE juce::juce_audio_utils Threads::Threads
  PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_binary_data(DistortionAssets SOURCES Assets/Fonts/Inter-Regular.ttf Assets/Fonts/Inter-Medium.ttf)
set_target_properties(DistortionAssets PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
target_link_libraries(Distortion PUBLIC DistortionAssets)
foreach(format VST3 AU AAX)
  if(TARGET Distortion_${format})
    add_custom_command(TARGET Distortion_${format} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:Distortion_${format}>/../Resources"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "$<TARGET_FILE_DIR:Distortion_${format}>/../Resources/Inter-OFL.txt" VERBATIM)
  endif()
endforeach()
if(DISTORTION_ENABLE_CLAP)
  include(FetchContent)
  FetchContent_Declare(clap_juce_extensions
    GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git
    GIT_TAG 7adee3a1bd4684d4caa5601100e364abccff4b4d GIT_SUBMODULES_RECURSE TRUE)
  FetchContent_MakeAvailable(clap_juce_extensions)
  clap_juce_extensions_plugin(TARGET Distortion CLAP_ID "ru.rainlinemusic.distortion"
    CLAP_FEATURES audio-effect analyzer stereo mono)
  if(APPLE)
    set(clap_license_dir "$<TARGET_FILE_DIR:Distortion_CLAP>/../Resources/Licenses")
  else()
    set(clap_license_dir "$<TARGET_FILE_DIR:Distortion_CLAP>/Licenses")
  endif()
  add_custom_command(TARGET Distortion_CLAP POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${clap_license_dir}"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Licenses" "${clap_license_dir}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "${clap_license_dir}/Inter-OFL.txt" VERBATIM)
  add_executable(DistortionCLAPTest Tests/clap_smoke.cpp)
  target_include_directories(DistortionCLAPTest PRIVATE "${clap_juce_extensions_SOURCE_DIR}/clap-libs/clap/include")
  target_link_libraries(DistortionCLAPTest PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(DistortionCLAPTest Distortion_CLAP)
  add_test(NAME CLAPSmoke COMMAND DistortionCLAPTest "$<TARGET_FILE:Distortion_CLAP>")
endif()
if(DISTORTION_BUILD_UI_TESTS)
  juce_add_console_app(DistortionIntegrationTest PRODUCT_NAME "Distortion integration tests")
  juce_generate_juce_header(DistortionIntegrationTest)
  target_sources(DistortionIntegrationTest PRIVATE Tests/integration_tests.cpp)
  target_include_directories(DistortionIntegrationTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/Distortion_artefacts/JuceLibraryCode")
  target_compile_definitions(DistortionIntegrationTest PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
  target_link_libraries(DistortionIntegrationTest PRIVATE Distortion juce::juce_audio_utils)
  add_test(NAME Integration COMMAND DistortionIntegrationTest "${CMAKE_CURRENT_BINARY_DIR}/ui-captures")
endif()
